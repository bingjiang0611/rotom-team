// Usage: copy to test/zz-probe.ts, run: node test/zz-probe.ts <hist-out-dir> <sentinel-file> off|on|worker. Set KEEP=<abs dir> to keep the data dir (e.g. reopen it later with older code for a rollback check).
import assert from "node:assert/strict";
import { cpSync, readFileSync, existsSync, rmSync, mkdtempSync, realpathSync, chmodSync } from "node:fs";
import { join } from "node:path";
import { tmpdir } from "node:os";
import { BACKGROUND_CONTEXT as context } from "@earendil-works/chord/context";
import { openNodeSqliteStorage } from "@earendil-works/pi-durable/storage/sqlite/node";
import { PocketApp } from "../src/server/app.ts";

const [hist, sentinel, mode] = process.argv.slice(2);
const kernel = mode !== "off";
const storageWorker = mode === "worker";
const keep = process.env.KEEP;
const dir = keep ?? realpathSync(mkdtempSync(join(tmpdir(), "d9a-probe-")));
const data = join(dir, "data");
if (!existsSync(data)) { cpSync(join(hist, "data"), data, { recursive: true }); chmodSync(data, 0o700); }
const work = mkdtempSync(join(tmpdir(), "d9a-work-"));
const before = readFileSync(sentinel, "utf8");
const id = 3 as never;
const later = () => Date.now() + 2 * 3600_000;

async function scan(method: string, q: object) {
  const s = await openNodeSqliteStorage(join(data, "pocket.sqlite"));
  const rows: any[] = [];
  let cursor;
  do { const p = await (s as any)[method](q, 128, cursor, context); rows.push(...p.items); cursor = p.next; } while (cursor !== undefined);
  await s.close(context);
  return rows;
}
async function open() {
  return PocketApp.open({ dataDir: data, defaultCwd: work, supervised: false, log: () => {}, kernel, storageWorker, now: later } as any);
}
const result: Record<string, unknown> = { mode };
for (const round of [1, 2]) {
  const app: any = await open();
  assert.equal(app.loader, undefined, "loader absent");
  await new Promise((r) => setTimeout(r, 4000));
  const u = app.config.users.find((user: any) => user.role === "owner");
  const md = (await app.transcripts.exportMarkdown(id, u)).markdown;
  result[`r${round}.exportHasHistory`] = md.includes("PROBE-HISTORY") && md.includes("old reply");
  try { await app.commands.submit(id, u, { text: "new work", requestId: `x${round}` }); result[`r${round}.submit`] = "accepted!"; }
  catch (e: any) { result[`r${round}.submit`] = `${e.status} ${e.message}`; }
  await app.close();
  const entries = await scan("scanEntries", { conversationId: id });
  result[`r${round}.archiveNotes`] = entries.filter((e) => e.data?.archive === true).length;
  result[`r${round}.scheduledSent`] = entries.filter((e) => JSON.stringify(e).includes("PROBE-SCHEDULED") && e.kind !== "pocket.note").length;
  result[`r${round}.newWork`] = entries.some((e) => JSON.stringify(e).includes("new work"));
  const tasks = [];
  for (const status of ["pending", "running", "waiting", "completing"]) tasks.push(...(await scan("scanTasks", { conversationId: id, status })));
  result[`r${round}.activeTasks`] = tasks.length;
  const subs = [];
  for (const status of ["queued", "placed"]) subs.push(...(await scan("scanSubmissions", { conversationId: id, status })));
  result[`r${round}.openSubmissions`] = subs.length;
  result[`r${round}.locks`] = ["harness.lock", "kernel.lock"].filter((f) => existsSync(join(data, f)));
}
result.sentinelUnchanged = readFileSync(sentinel, "utf8") === before;
console.log(JSON.stringify(result));
if (!keep) rmSync(dir, { recursive: true, force: true });
process.exit(0);
